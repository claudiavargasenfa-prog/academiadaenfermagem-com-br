let diagnosticosAtuais = [];

function atualizarDiagnosticos() {
  const checks = document.querySelectorAll(".symptom:checked");
  const codes = new Set();
  checks.forEach(cb => {
    (mapa[cb.value] || []).forEach(c => codes.add(c));
  });
  diagnosticosAtuais = Array.from(codes).sort();

  const container = document.getElementById("diagnosisContainer");
  const counter = document.getElementById("diagnosisCounter");
  const n = diagnosticosAtuais.length;
  counter.textContent = n + (n === 1 ? " diagnóstico identificado" : " diagnósticos identificados");

  // Progresso
  document.querySelectorAll("#step1, #step2, #step3, #step4").forEach(s => {
    s.querySelector("div:first-child").style.background = "#e0e0e0";
    s.querySelector("div:first-child").style.color = "#757575";
    s.querySelector("div:first-child").style.transform = "scale(1)";
    s.querySelector("div:first-child").style.boxShadow = "0 3px 10px rgba(0,0,0,0.1)";
  });
  document.querySelector("#step1 div:first-child").style.background = "#1a6b3a";
  document.querySelector("#step1 div:first-child").style.color = "#fff";
  document.querySelector("#step1 div:first-child").style.transform = "scale(1.1)";
  if (n > 0) {
    document.querySelector("#step2 div:first-child").style.background = "#1a6b3a";
    document.querySelector("#step2 div:first-child").style.color = "#fff";
    document.querySelector("#step2 div:first-child").style.transform = "scale(1.1)";
  }

  if (n === 0) {
    container.innerHTML = '<div style="text-align:center;padding:40px 20px;color:#555;"><div style="font-size:48px;margin-bottom:12px;">🔍</div><p>Selecione sinais e sintomas no exame físico para visualizar os diagnósticos sugeridos.</p></div>';
    return;
  }

  let html = '<div style="display:grid;grid-template-columns:repeat(auto-fit,minmax(280px,1fr));gap:16px;">';
  diagnosticosAtuais.forEach(c => {
    const d = db[c];
    if (d) {
      html += '<div style="background:#fff;border-radius:12px;padding:18px;border-left:5px solid #1a6b3a;box-shadow:0 3px 12px rgba(0,0,0,0.06);">';
      html += '<span style="display:inline-block;background:#fff3e0;color:#e65100;padding:3px 10px;border-radius:20px;font-size:12px;font-weight:700;margin-bottom:8px;">NANDA ' + c + '</span>';
      html += '<h4 style="font-size:17px;font-weight:700;color:#0d3b1e;margin:0 0 10px 0;">' + d.name + '</h4>';
      html += '<div style="font-size:13px;color:#555;margin-bottom:4px;"><strong style="color:#1a6b3a;">NOC:</strong> ' + d.noc + '</div>';
      html += '<div style="font-size:13px;color:#555;"><strong style="color:#1a6b3a;">NIC:</strong> ' + d.nic + '</div>';
      html += '</div>';
    }
  });
  html += '</div>';
  container.innerHTML = html;
}

function formatarData() {
  const d = new Date();
  return d.toLocaleDateString("pt-BR", { day: "2-digit", month: "2-digit", year: "numeric" });
}

function gerarPrescricao() {
  if (diagnosticosAtuais.length === 0) {
    alert("Selecione pelo menos um sinal ou sintoma para gerar a prescrição.");
    return;
  }

  const nome = document.getElementById("nomePaciente").value || "______________________________________";
  const idade = document.getElementById("idadePaciente").value || "__________";
  const leito = document.getElementById("leitoPaciente").value || "_______";
  const diagMed = document.getElementById("diagMedico").value || "_______________________________";

  let texto = "SERVIÇO DE ENFERMAGEM\n";
  texto += "PRESCRIÇÃO DE ENFERMAGEM\n";
  texto += "Data: " + formatarData() + "\n\n";
  texto += "Paciente: " + nome + "          Idade: " + idade + "\n";
  texto += "Leito: " + leito + "          Diagnóstico Médico: " + diagMed + "\n";
  texto += "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n\n";

  diagnosticosAtuais.forEach((c, i) => {
    const d = db[c];
    const iv = interv[c] || ["Avaliar conforme protocolo institucional;"];
    if (d) {
      texto += (i + 1) + ". DE: " + c + " — " + d.name + "\n";
      texto += "   NOC: " + d.noc + "\n";
      texto += "   NIC: " + d.nic + "\n\n";
      texto += "   PRESCRIÇÕES:\n";
      iv.forEach(x => { texto += "   - " + x + "\n"; });
      texto += "   Frequência: De acordo com evolução clínica e protocolo.\n";
      texto += "\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n\n";
    }
  });

  texto += "PRESCRIÇÕES GERAIS:\n";
  texto += "- Monitorar sinais vitais de 4/4h;\n";
  texto += "- Higiene e conforto a cada turno;\n";
  texto += "- Mudança de decúbito a cada 2h quando acamado;\n";
  texto += "- Orientar paciente/acompanhante sobre cuidados;\n";
  texto += "- Registrar todas as intervenções no prontuário.\n\n";
  texto += "Assinatura do Enfermeiro: _________________________________\n";
  texto += "COREN: _________________________\n";

  document.getElementById("prescriptionText").textContent = texto;

  // progresso
  document.querySelectorAll("#step1 div:first-child, #step2 div:first-child, #step3 div:first-child, #step4 div:first-child").forEach(s => {
    s.style.background = "#1a6b3a";
    s.style.color = "#fff";
    s.style.transform = "scale(1)";
  });
  document.querySelector("#step3 div:first-child").style.transform = "scale(1.1)";
}
